# 报错信息点击复制实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给两处被展示层截断的报错信息（MCP 列表失败原因、运行指示器重试错误详情）加「点击文字复制完整原文」能力，复制后短暂显示「已复制 ✓」。

**Architecture:** 零新增文件（生产代码），全部为既有两类的小改动：`RunningIndicator` 抽 `renderText()` 统一文案重绘口径 + 点击复制 + `PauseTransition` 反馈计时；`SettingsDialog` MCP 列表 cell 的失败原因 `Label` 加点击复制。完整原文均来自既有数据载体的原字段（`RetryProgress.body` / `McpServer.failReason`），不减不截断。

**Tech Stack:** Java 8、JavaFX 8（JDK 自带 jfxrt）、Maven、JUnit 4。

**设计文档:** `docs/superpowers/specs/2026-09-17-报错信息点击复制-design.md`（已获批，提交 `c31db3a`）

## Global Constraints

- **JDK 8 语法**（禁 `var`/`List.of`/文本块/lambda 之外的 Java 9+ API）；注释、提交信息均中文（commit 用 conventional 格式）。
- **复制内容逐字定稿**：指示器复制 `RetryProgress.body` **完整原文**；MCP 复制 `McpServer.failReason` **完整原文**。均不截断、不加前缀、不剥离前缀。
- **反馈文案逐字定稿**：`已复制 ✓`（`COPIED_TEXT`），显示时长 `1500` ms（`COPY_FEEDBACK_MS`）。
- **纯逻辑函数签名定稿**（两个类各自 package-private static，供单测）：
  - `RunningIndicator.copyText(RetryProgress p)`：`p == null || p.body == null || p.body.trim().isEmpty()` → `null`，否则返回 `p.body`。
  - `SettingsDialog.copyableFailReason(McpServer s)`：`s == null || s.state != FAILED || s.failReason == null || s.failReason.trim().isEmpty()` → `null`，否则返回 `s.failReason`（状态判断入函数，与设计「仅 FAILED 可点」一致且可单测）。
- **不改**：`shorten`（40 字符）与 `BODY_MAX_CHARS`（200 字符）展示截断；聊天区【系统】错误行；`LlmException`/`RetryProgress`/`McpServer` 数据载体；主题 CSS。
- **字体/光标**：可复制时才设 `Cursor.HAND`，不可复制时为默认光标（`setCursor(null)`）。
- **动画规约**：组件隐藏前必须停全部计时器（`stopAnimations` 内新增停 `copyFeedback`），防引用泄漏与过期回调。
- 每任务结束：相关测试全绿并提交；最后 `mvn test` 全量 + `mvn -q package -DskipTests`。
- JavaFX 组件本体不做 UI 自动化单测（沿用现状），交互由 Task 3 探针 + 真机手动验证。

```java
     *  复制后文字短暂显示「已复制 ✓」，到期按当前状态重绘 */
    void copyErrorDetail() {
        String full = copyText(retryProgress);
        if (full == null) return;
        ClipboardContent cc = new ClipboardContent();
        cc.putString(full);
        Clipboard.getSystemClipboard().setContent(cc);
        if (copyFeedback != null) copyFeedback.stop();
        text.setText(COPIED_TEXT);
        copyFeedback = new PauseTransition(Duration.millis(COPY_FEEDBACK_MS));
        copyFeedback.setOnFinished(e -> { copyFeedback = null; renderText(); });
        copyFeedback.play();
    }

    /** 按当前状态重绘文案（统一口径：状态变化/反馈到期共用）：重试态 → 重试文案；
     *  否则压缩/轮换文案。反馈计时未到期即被新状态覆盖时在此停表（提示让位于实时状态） */
    private void renderText() {
        if (copyFeedback != null) { copyFeedback.stop(); copyFeedback = null; }
        text.setText(retryBase != null ? retryText(retryProgress, retryBase)
                : displayText(compressing, pickText(rnd)));
    }
```

```java
package com.minion.gui;

import com.minion.core.agent.RetryProgress;
import javafx.animation.PauseTransition;
import javafx.application.Application;
import javafx.application.Platform;
import javafx.scene.Node;
import javafx.scene.Scene;
import javafx.scene.control.Label;
import javafx.scene.input.Clipboard;
import javafx.scene.layout.StackPane;
import javafx.stage.Stage;
import javafx.util.Duration;

/** 报错提示点击复制探针（临时，不入库）：点击复制完整 body + 「已复制 ✓」反馈 + 1.5s 恢复 */
public class RunningIndicatorCopyProbe extends Application {

    private static int failed = 0;

    public static void main(String[] args) { launch(args); }

    @Override public void start(Stage st) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < 300; i++) sb.append((char) ('a' + i % 26));
        final String body = sb.toString();
        final RunningIndicator ind = new RunningIndicator();
        ind.setRunning(true); // 必须先置 running：setRetryProgress 对非运行态直接 return（早退拦截）
        ind.setRetryProgress(RetryProgress.of(2, 500, body)); // 300 字符 body，远超展示截断 200
        st.setScene(new Scene(new StackPane(ind), 420, 100));
        st.setTitle("copy-probe");
        st.show();

        Platform.runLater(() -> {
            ind.copyErrorDetail(); // 与点击事件同一入口（copyErrorDetail 为 package-private）
            check("剪贴板 = 完整 body（未截断）", body.equals(Clipboard.getSystemClipboard().getString()));
            check("反馈文案 = 已复制 ✓", RunningIndicator.COPIED_TEXT.equals(labelText(ind)));
            PauseTransition wait = new PauseTransition(Duration.millis(1800));
            wait.setOnFinished(e -> {
                String t = labelText(ind);
                check("1.5s 后恢复重试文案", t != null && t.contains("500服务报错") && !t.contains("已复制"));
                System.out.println(failed == 0 ? "PROBE PASS" : "PROBE FAIL: " + failed);
                Platform.exit();
                System.exit(failed == 0 ? 0 : 1);
            });
            wait.play();
        });
    }

    private static String labelText(RunningIndicator ind) {
        for (Node n : ind.getChildrenUnmodifiable()) if (n instanceof Label) return ((Label) n).getText();
        return null;
    }

    private static void check(String what, boolean ok) {
        if (!ok) failed++;
        System.out.println((ok ? "PASS " : "FAIL ") + what);
    }
}
```
